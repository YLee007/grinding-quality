cmake_minimum_required(VERSION 3.20)
project(grinding_quality LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
add_executable(grinding_quality src/main.cpp src/quality.cpp)
target_include_directories(grinding_quality PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(grinding_quality PRIVATE ${OpenCV_LIBS} yaml-cpp nlohmann_json::nlohmann_json)
target_compile_options(grinding_quality PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(grinding_quality PRIVATE -ffp-contract=off)
endif()
install(TARGETS grinding_quality RUNTIME DESTINATION bin)
